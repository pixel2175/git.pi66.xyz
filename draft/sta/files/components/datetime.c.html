#include <stdio.h>
#include <time.h>

#include "../util.h"

const char *
datetime(const char *fmt)
{
    static _Thread_local char buf[1024];
    time_t t;

    t = time(NULL);
    if (!strftime(buf, sizeof(buf), fmt, localtime(&t))) {
        warn("strftime: Result string exceeds buffer size");
        return NULL;
    }

    return buf;
}
